cmake_minimum_required(VERSION 3.20)
project(hdmaxdec VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/utf-8)
endif()
option(HDMAXDEC_FFMPEG "Build the optional libavutil frame adapter" OFF)
option(HDMAXDEC_TESTS "Build decoder regression tests" ON)
add_library(hdmaxdec SHARED src/core.cpp src/xll.cpp src/decoder.cpp src/layout.cpp src/file.cpp src/playback.cpp src/export.cpp)
target_include_directories(hdmaxdec PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_definitions(hdmaxdec PRIVATE HDMAXDEC_BUILD PUBLIC HDMAXDEC_SHARED)
if(MSVC)
  target_compile_options(hdmaxdec PRIVATE /W4 /permissive-)
else()
  target_compile_options(hdmaxdec PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(APPLE)
  target_link_libraries(hdmaxdec PRIVATE "-framework AudioToolbox" "-framework CoreAudio" "-framework CoreFoundation")
elseif(WIN32)
  target_link_libraries(hdmaxdec PRIVATE ole32 uuid)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  find_package(ALSA REQUIRED)
  target_link_libraries(hdmaxdec PRIVATE ALSA::ALSA)
endif()
add_executable(hdmaxdec_cli cli/main.cpp)
if(APPLE)
  target_sources(hdmaxdec_cli PRIVATE cli/macos/qc_engine.cpp cli/macos/qc_player.cpp)
  target_link_libraries(hdmaxdec_cli PRIVATE "-framework AppKit" "-framework CoreGraphics" "-framework UniformTypeIdentifiers" objc)
  add_custom_command(TARGET hdmaxdec_cli POST_BUILD
    COMMAND sh ${CMAKE_CURRENT_SOURCE_DIR}/cli/macos/package.sh $<TARGET_FILE_DIR:hdmaxdec_cli> $<TARGET_FILE:hdmaxdec_cli> ${CMAKE_CURRENT_SOURCE_DIR}
    VERBATIM)
endif()
set_target_properties(hdmaxdec_cli PROPERTIES OUTPUT_NAME hdmaxdec)
target_link_libraries(hdmaxdec_cli PRIVATE hdmaxdec)
if(MINGW)
  target_link_options(hdmaxdec_cli PRIVATE -municode)
endif()
if(HDMAXDEC_FFMPEG)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(AVUTIL REQUIRED IMPORTED_TARGET libavutil>=58)
  add_library(hdmaxdec_avframe ffmpeg/hdmaxdec_avframe.c)
  target_include_directories(hdmaxdec_avframe PUBLIC ffmpeg)
  target_link_libraries(hdmaxdec_avframe PUBLIC hdmaxdec PkgConfig::AVUTIL)
endif()
if(HDMAXDEC_TESTS)
  enable_testing()
  add_executable(hdmaxdec_tests tests/regression.cpp)
  target_include_directories(hdmaxdec_tests PRIVATE src)
  target_link_libraries(hdmaxdec_tests PRIVATE hdmaxdec)
  add_test(NAME decoder_regression COMMAND hdmaxdec_tests)
  add_executable(hdmaxdec_export_tests tests/export.cpp)
  target_link_libraries(hdmaxdec_export_tests PRIVATE hdmaxdec)
  add_test(NAME pcm_export COMMAND hdmaxdec_export_tests)
  if(HDMAXDEC_FFMPEG)
    add_executable(hdmaxdec_ffmpeg_tests tests/ffmpeg.cpp)
    target_link_libraries(hdmaxdec_ffmpeg_tests PRIVATE hdmaxdec_avframe)
    add_test(NAME ffmpeg_frame_adapter COMMAND hdmaxdec_ffmpeg_tests)
  endif()
endif()
include(GNUInstallDirs)
install(TARGETS hdmaxdec hdmaxdec_cli RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES include/hdmaxdec.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES
  "${CMAKE_CURRENT_SOURCE_DIR}/../LICENSES/hdmaxdec-LGPL-2.1-or-later.txt"
  "${CMAKE_CURRENT_SOURCE_DIR}/../LICENSES/LGPL-2.1.txt"
  "${CMAKE_CURRENT_SOURCE_DIR}/../LICENSES/GPL-2.0.txt"
  DESTINATION ${CMAKE_INSTALL_DATADIR}/hdmaxdec/LICENSES)
